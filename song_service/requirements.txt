# Song service packages: ACE-Step (music) and the lyric model.
#
# install.py is the authoritative installer, not this file. It installs these same
# specs (it reads them from models.json), then downloads ~10 GB of weights into
# song_service/models/ — which a requirements file cannot express:
#
#   song_service\.venv\Scripts\python.exe song_service\install.py            packages + weights
#   song_service\.venv\Scripts\python.exe song_service\install.py --packages  code only
#   song_service\.venv\Scripts\python.exe song_service\install.py --check     report, change nothing
#
# This file exists so the package list is readable and machine-consumable without
# opening models.json, and so a manual install has the pins in one place:
#
#   uv pip install --python song_service\.venv\Scripts\python.exe --extra-index-url https://download.pytorch.org/whl/cu126 torch==2.14.1+cu126 torchaudio==2.14.1+cu126
#   uv pip install --python song_service\.venv\Scripts\python.exe -r song_service\requirements.txt
#
# The +cu126 pin matters for the same reason it does in voice_service: PyPI's
# Windows torch wheel is the CPU build, and a CPU build cannot render a song in
# seconds.

# The music model. Without the +cu126 wheel this resolves to the CPU build.
torch==2.14.1+cu126
torchaudio==2.14.1+cu126

# 5.x needs huggingface-hub 2.x, which the ACE install does not use.
transformers==4.57.6
huggingface-hub==0.36.2

# The model's own code; the 7 GB checkpoint is downloaded by install.py.
git+https://github.com/ace-step/ACE-Step.git

flask
flask-cors
hf_transfer
soundfile
# librosa: ACE-Step's audio loading, and the pitch shift that maps the sung vocal
# into the song's key. scipy: the low-pass and room half of that map - librosa
# pulls it in, but the code imports scipy.signal by name.
librosa
scipy
# demucs: splits a rendered song into the model's vocal and its backing, so the
# voice map is applied to the words and not to the instruments.
demucs
# numpy: app.py imports it by name for the mix and the pitch measurement. Pinned to
# the version the models run on here, so a reinstall cannot move the array API
# under torch.
numpy==2.4.6
# No pyworld here on purpose: the WORLD vocoder held the formants but its full
# resynthesis measured +281% spectral flatness on a separated vocal - a grain you
# can hear. The shift is a phase vocoder followed by a formant-envelope correction
# instead, which needs nothing beyond librosa/numpy.
