# Voice-cloning environment: sings the AI's lyrics in a recorded voice.
#
# Install with install.py, not with this file directly:
#
#   voice_service\.venv\Scripts\python.exe voice_service\install.py --check
#   voice_service\.venv\Scripts\python.exe voice_service\install.py
#
# This file is the package list the script installs. It cannot express the part
# that matters: chatterbox-tts pins torch==2.6.0, and PyPI's wheel for that version
# is the CPU build. A plain `pip install -r requirements.txt` therefore replaces a
# working CUDA torch with one that reports cuda False - silently, with no error -
# and the service then runs on the processor, minutes per phrase instead of
# seconds. install.py installs these packages and then restores the +cu126 wheels
# of the same version, which keeps chatterbox's pin and the card in use.
#
# Installing one of the small packages on its own later does not touch torch:
#
#   uv pip install --python voice_service\.venv\Scripts\python.exe soundfile
#
# Check the card at any time with install.py --check, or:
#
#   voice_service\.venv\Scripts\python.exe -c "import torch; print(torch.__version__, torch.cuda.is_available())"
#
# which must print 2.6.0+cu126 True.

chatterbox-tts==0.1.7
# chatterbox's own pin is numpy<2; 2.x breaks its audio helpers at import.
numpy==1.26.4
# perth (chatterbox's watermarker) imports pkg_resources, which setuptools 81+
# removed. Without this the watermarker is None and from_pretrained dies with
# "TypeError: 'NoneType' object is not callable".
setuptools<81
